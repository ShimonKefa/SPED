[SQL]
1 = SELECT 'E111' REG, MAX(COD_AJ) AS COD_AJ, '' DESCR_COMPL_AJ, SUM(VL_ICMS) VL_ICMS, COD_APU FROM (
2 = SELECT DISTINCT DOC.COD_DOC, 0 COD_DOCIT, DOC.MOV_DOC, COALESCE(APUTRIB.CICMSSTFR_APU, 'ES99991993') COD_AJ, DOC.OBSLIV_DOC DESCR_COMPL_AJ,
3 = '' COD_ITEM, COALESCE(DOC.BCAUTO_DOC,0) VL_BC_ICMS, COALESCE(DOC.ALIQAUTO_DOC,0) ALIQ_ICMS, COALESCE(ICMSAUTO_DOC,0) VL_ICMS, 0.0 VL_OUTROS, APUTRIB.COD_APU FROM DOCIT
4 = INNER JOIN DOC ON (DOC.COD_DOC = DOCIT.COD_DOC)
5 = INNER JOIN ESTAB ON (ESTAB.COD_ESTAB = DOC.COD_ESTAB)
6 = INNER JOIN CFOP ON (CFOP.COD_CFOP = DOCIT.CFOP_DOCIT AND CFOP.INT_CFOP = DOCIT.INTCFOP_DOCIT)
7 = LEFT JOIN PROD ON (PROD.COD_ESTQ = DOCIT.COD_ESTQ)
8 = LEFT JOIN GRPIT ON (PROD.COD_GRP = GRPIT.COD_GRP AND ESTAB.COD_ESTAB = GRPIT.COD_ESTAB AND CFOP.CLCTB_CFOP = GRPIT.CLCTB_CFOP), APUTRIB 
9 = WHERE COALESCE(DOC.CANC_DOC,'N') <> 'S' 
10 = AND DOC.COD_AIDF > 0 AND COALESCE(ICMSAUTO_DOC,0) > 0 AND COALESCE(DOC.MODFRETE2_DOC, "I") in ("I","E") AND DOC.PRINT_DOC = 'S' AND DOC.COD_MOD IN ('1', '1B', '04', '4', '55')
11 = AND DOC.COD_ESTAB = :COD_ESTAB AND (DOC.EMISS_DOC >= :Data_Inicial || ' 00:00:00' AND DOC.EMISS_DOC <= :Data_Final || ' 23:59:59' AND DOC.MOV_DOC = 'S') AND 
12 = (APUTRIB.TRIB_APU = "ICMS" AND APUTRIB.DATA1_APU >= :Data_Inicial AND APUTRIB.DATA2_APU <= :Data_Final AND APUTRIB.COD_ESTAB = :COD_ESTAB)
13 = ) GROUP BY COD_APU
